@ClientName VARCHAR(100),
@ClientLocation VARCHAR (150),
@Duration NVARCHAR(100),
@Ammount DECIMAL,

AS
BEGIN

 SET NOCOUNT ON;

		 ClientLocation = @ClientLocation,
		 Duration = @Duration,
		 Ammount = @Ammount,
		 ProjectCategory = @ProjectCategory,
		 DeadLine = @DeadLine
	WHERE ProjectId =  @ProjectId
   END

   ELSE
   BEGIN
     INSERT INTO Projects ( 
	   ProjectName ,
	   ProjectDate,
	   ClientName,
	   ClientLocation,
	   Duration,
	   Ammount,
	   ProjectCategory,
	   DeadLine
	 )
	 VALUES (
	 @ProjectName,
	 @ProjectDate,
	 @ClientName,
	 @ClientLocation,
	 @Duration,
	 @Ammount,
	 @ProjectCategory,
	 @DeadLine
	 )
   END

-- Lightweight login lookup — same shape as your existing AuthLogin_Sp but
-- with ProfilePicture forced to NULL. Login only needs the password hash;
-- there's no reason to transfer a large base64 image on every sign-in.
-- Adjust the SELECT list if your EmploayeeAuth columns differ slightly.
CREATE   PROCEDURE [dbo].[AuthLoginLite_Sp]
    @AuthName NVARCHAR(100)
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        Id,
        AuthName,
        AuthRole,
        PasswordHash,
        Country,
        Age,
        Gender,
        Salary,
        JoiningDate,
        CAST(NULL AS NVARCHAR(MAX)) AS ProfilePicture,
        Email
    FROM dbo.EmploayeeAuth
    WHERE AuthName = @AuthName;
END
GO

-- New upsert proc (INSERT when @id = 0, otherwise UPDATE) used by both
-- Add Employee and Edit Employee — replaces the old Employee_InsertSp
-- call in the C# repository; the original proc is left untouched.
CREATE   PROCEDURE [dbo].[Employee_UpsertSp]
    @id INT,
    @EmpName NVARCHAR(100) = NULL,
    @EmpAge INT = NULL,
    @EmpGender NVARCHAR(50) = NULL,
    @EmpSalary DECIMAL(18, 0) = NULL,
    @EmpCountry NVARCHAR(50) = NULL,
    @EmpEmail NVARCHAR(500) = NULL,
    @ReportingManagerId INT = NULL,
    @TeamLeadId INT = NULL
AS
BEGIN
    SET NOCOUNT ON;

    IF @id IS NULL OR @id = 0
    BEGIN
        INSERT INTO dbo.EmployeeData
            (EmpName, EmpAge, EmpGender, EmpSalary, EmpCountry, EmpEmail, ReportingManagerId, TeamLeadId)
        VALUES
            (@EmpName, @EmpAge, @EmpGender, @EmpSalary, @EmpCountry, @EmpEmail, @ReportingManagerId, @TeamLeadId);
    END
    ELSE
    BEGIN
        UPDATE dbo.EmployeeData
        SET EmpName = @EmpName,
            EmpAge = @EmpAge,
            EmpGender = @EmpGender,
            EmpSalary = @EmpSalary,
            EmpCountry = @EmpCountry,
            EmpEmail = @EmpEmail,
            ReportingManagerId = @ReportingManagerId,
            TeamLeadId = @TeamLeadId

    END
END

CREATE PROCEDURE [dbo].[EmployeeData_SP]
 @ACTION VARCHAR(50),
 @id INT,
 @EmpName VARCHAR (100),
 @EmpAge INT ,
 @EmpGender VARCHAR (50),
 @EmpSalary DECIMAL,
 @EmpCountry VARCHAR (50) ,
 @EmpEmail NVARCHAR(500)
AS
BEGIN
   
   IF( @ACTION = 'GETALL')
   BEGIN
    SELECT * FROM EmployeeData
   END 

   ELSE IF( @ACTION = 'GETBYID' )
   BEGIN
    SELECT * FROM EmployeeData
	WHERE id = @id
   END

   ELSE IF ( @ACTION = 'INSERT' )
   BEGIN
    INSERT INTO EmployeeData ( EmpName ,  EmpAge , EmpGender ,  EmpSalary ,  EmpCountry , EmpEmail)
	VALUES
	( @EmpName , @EmpAge , @EmpGender , @EmpSalary , @EmpCountry , @EmpEmail )
   END


   ELSE IF ( @ACTION = 'UPDATE' )
   BEGIN
     UPDATE EmployeeData
	 SET EmpName = @EmpName,
	     EmpAge = @EmpAge,
		 EmpGender = @EmpGender ,
		 EmpSalary = @EmpSalary,
		 EmpCountry = @EmpCountry ,
		 EmpEmail = @EmpEmail
     WHERE id = @id
   END


   ELSE IF ( @ACTION = 'DELETE' )
   BEGIN
     DELETE EmployeeData
	 WHERE id = @id
   END
END
GO

-- Update the editable profile fields for a user. Pass NULL for
-- any field you don't want to change; the COALESCE keeps the
-- existing value in that case.
CREATE   PROCEDURE [dbo].[UpdateProfile_Sp]
    @AuthName NVARCHAR(100),
    @AuthRole NVARCHAR(50),
    @NewAuthName NVARCHAR(100) = NULL,
    @Email NVARCHAR(255) = NULL,
    @Country NVARCHAR(100) = NULL,
    @Age INT = NULL,
    @Gender NVARCHAR(100) = NULL,
    @ProfilePicture NVARCHAR(MAX) = NULL
AS
BEGIN
    SET NOCOUNT ON;

    UPDATE dbo.EmploayeeAuth
    SET
        AuthName = COALESCE(@NewAuthName, AuthName),
        Email = COALESCE(@Email, Email),
        Country = COALESCE(@Country, Country),
        Age = COALESCE(@Age, Age),
